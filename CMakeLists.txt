cmake_minimum_required(VERSION 3.20)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
project(rt_androidauto VERSION 0.1.0 LANGUAGES C CXX)
option(RTAA_TESTS "Build production-source host tests" ON)
option(RTAA_SANITIZERS "Enable ASan and UBSan (Clang/GCC)" OFF)
option(RTAA_EMBEDDED_TLS_CONFIG "Use the portable TLS client-only configuration" OFF)
set(RTAA_RTTHREAD_ROOT "" CACHE PATH "Optional RT-Thread headers for its real port")
set(RTAA_RTTHREAD_CONFIG "" CACHE PATH "Directory containing the product rtconfig.h")
set(RTAA_MBEDTLS_SOURCE "" CACHE PATH "mbedTLS 2.28.9 source checkout")
if(RTAA_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(RTAA_MBEDTLS_SOURCE)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${RTAA_MBEDTLS_SOURCE}" mbedtls EXCLUDE_FROM_ALL)
  if(RTAA_EMBEDDED_TLS_CONFIG)
    foreach(lib mbedtls mbedx509 mbedcrypto)
      target_compile_definitions(${lib} PUBLIC MBEDTLS_CONFIG_FILE="rtaa_mbedtls.h")
      target_include_directories(${lib} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/config")
    endforeach()
  endif()
else()
  find_path(MBEDTLS_INCLUDE_DIR mbedtls/ssl.h REQUIRED)
  foreach(lib mbedtls mbedx509 mbedcrypto)
    find_library(${lib}_PATH ${lib} REQUIRED)
    add_library(${lib} UNKNOWN IMPORTED)
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${${lib}_PATH}"
      INTERFACE_INCLUDE_DIRECTORIES "${MBEDTLS_INCLUDE_DIR}")
  endforeach()
endif()
add_library(rtaa STATIC src/framing.c src/aoa.c src/transport.c src/tls_mbedtls.c src/connection.c
  src/receiver.c generated/aap.pb.c third_party/nanopb/pb_common.c
  third_party/nanopb/pb_encode.c third_party/nanopb/pb_decode.c)
target_compile_features(rtaa PUBLIC c_std_11)
target_include_directories(rtaa PUBLIC include PRIVATE generated third_party/nanopb)
target_link_libraries(rtaa PUBLIC mbedtls mbedx509 mbedcrypto)
if(RTAA_RTTHREAD_ROOT)
  if(NOT EXISTS "${RTAA_RTTHREAD_CONFIG}/rtconfig.h")
    message(FATAL_ERROR "RTAA_RTTHREAD_CONFIG must contain the actual product rtconfig.h")
  endif()
  target_sources(rtaa PRIVATE port/rtthread/rtaa_rtthread.c)
  target_include_directories(rtaa PRIVATE "${RTAA_RTTHREAD_ROOT}/include"
    "${RTAA_RTTHREAD_ROOT}/components/legacy"
    "${RTAA_RTTHREAD_ROOT}/components/finsh"
    "${RTAA_RTTHREAD_ROOT}/components/drivers/include" "${RTAA_RTTHREAD_CONFIG}")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(rtaa PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(RTAA_TESTS)
  enable_testing()
  foreach(name core protobuf tls receiver)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE rtaa)
    target_include_directories(test_${name} PRIVATE generated third_party/nanopb tests)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  add_executable(test_rtthread tests/test_rtthread.c port/rtthread/rtaa_rtthread.c)
  target_include_directories(test_rtthread PRIVATE include tests/rtthread)
  add_test(NAME rtthread COMMAND test_rtthread)
  add_library(public_headers_c OBJECT tests/headers.c)
  target_include_directories(public_headers_c PRIVATE include)
  add_library(public_headers_cpp OBJECT tests/headers.cpp)
  target_include_directories(public_headers_cpp PRIVATE include)
endif()
